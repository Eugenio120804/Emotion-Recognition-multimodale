import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.metrics import confusion_matrix

# ==========================================
# 1. CONFIGURAZIONE ESPERIMENTO
# ==========================================
NOME_FILE = "risultati_EXP2_v2_Distribuzione.csv"
COLONNA_PREDIZIONE = "VLM_Prediction"  
TITOLO_GRAFICO = "Esperimento tramite distribuzione di probabilità di BERT"

# ==========================================
# 2. CARICAMENTO E PULIZIA DATI
# ==========================================
# Prova a leggere con il punto e virgola, se fallisce usa la virgola
df = pd.read_csv(NOME_FILE, sep=';')
if len(df.columns) == 1:
    df = pd.read_csv(NOME_FILE, sep=',')

# Rimuove spazi accidentali dai nomi delle colonne e dai dati
df.columns = df.columns.str.strip()
y_true = df['Ground_Truth'].astype(str).str.strip().str.lower()
y_pred = df[COLONNA_PREDIZIONE].astype(str).str.strip().str.lower()

# ==========================================
# 3. GENERAZIONE MATRICE
# ==========================================
# Etichette standard IEMOCAP nell'ordine corretto
all_labels = ['ang', 'dis', 'exc', 'fea', 'fru', 'hap', 'neu', 'oth', 'sad', 'sur']

# Mantiene solo le etichette effettivamente presenti nel file per pulizia grafica
present_labels = sorted(list(set(y_true.unique()) | set(y_pred.unique())))
etichette = [label for label in all_labels if label in present_labels]

cm = confusion_matrix(y_true, y_pred, labels=etichette)

# ==========================================
# 4. IMPOSTAZIONI GRAFICHE E SALVATAGGIO
# ==========================================
plt.figure(figsize=(10, 8))
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
            xticklabels=etichette, yticklabels=etichette, cbar=False)

plt.title(TITOLO_GRAFICO, fontsize=16, fontweight='bold', pad=15)
plt.xlabel('Predizione del Modello', fontsize=12)
plt.ylabel('Emozione Reale (Ground Truth)', fontsize=12)

# Salva l'immagine a 300 DPI (qualità stampa)
nome_output = NOME_FILE.replace('.csv', '_Matrice.png').replace('.xlsx', '_Matrice.png')
plt.tight_layout()
plt.savefig(nome_output, dpi=300, bbox_inches='tight')
print(f"Matrice di confusione generata e salvata come: {nome_output}")
